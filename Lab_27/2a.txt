#include <iostream>
using namespace std;

int main(void) {
int a = 8, b = 0, c = 0;
cout<<"Enter number for divide: ";
cin >> b;

try {
    if(b == 0) {
        
     throw(0);
     
    }
    c = a / b;
}

catch(int error){
    
    if(error==0){
        
        cout<<"Your input is not valid, you can't divide by zero."<<endl;
    }
    
 }
cout << c << endl;
return 0;
}
