#include <iostream>
using namespace std;

const int DivideByZero = 111;

float internaldiv(float arg1, float arg2) {
    if (0 == arg2)
        throw DivideByZero;
    return arg1 / arg2;
}

// Version 2: div checks the argument and lets the exception bubble up
float div(float arg1, float arg2) {
    if (arg2 == 0.0) {
        throw DivideByZero; // Forward/throw the exception manually
    }
    return internaldiv(arg1, arg2);
}

int main(void) {
    float a, b;
    while (cin >> a) {
        cin >> b;
        try {
            float r = div(a, b);
            cout << r << endl;
        }
        catch (int e) {
            if (e == DivideByZero) {
                cout << "Are you kidding me?" << endl;
                cout << "Your input is not valid. You can't divide by zero." << endl;
            }
        }
    }
    return 0;
}
